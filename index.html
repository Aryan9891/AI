import random

# 1. create_environment() - creates and returns the 2x1 array representing true hidden state (0 = Clean, 1 = Dirty)
def create_environment():
    return [0, 0]

# 2. place_dirt(env) - randomly places dirt (1) in cells of the grid
def place_dirt(env):
    for i in range(len(env)):
        env[i] = random.choice([0, 1])
    # Ensure at least one cell has dirt for a meaningful simulation
    if sum(env) == 0:
        env[random.randint(0, len(env) - 1)] = 1
    return env

# 3. place_agent() - randomly selects and returns a starting position (cell index) for the agent
def place_agent(grid_size=2):
    return random.randint(0, grid_size - 1)

# 4. create_knowledge() - creates and returns agent's knowledge array, starting as None (unknown)
def create_knowledge(grid_size=2):
    return [None] * grid_size

# 5. perceive(env, knowledge, pos) - lets agent sense true status of occupied cell, updating knowledge
def perceive(env, knowledge, pos):
    knowledge[pos] = env[pos]
    return knowledge

# 6. is_dirty(env, pos) - checks whether agent's current cell is actually dirty
def is_dirty(env, pos):
    return env[pos] == 1

# 7. vacuum(env, knowledge, pos) - cleans current cell in true env and updates knowledge to clean (0)
def vacuum(env, knowledge, pos):
    env[pos] = 0
    knowledge[pos] = 0
    return env, knowledge

# 8. get_possible_moves(pos, grid_size=2) - returns valid moves (Left/Right) based on boundaries
def get_possible_moves(pos, grid_size=2):
    moves = []
    if pos > 0:
        moves.append('Left')
    if pos < grid_size - 1:
        moves.append('Right')
    return moves

# 9. move_agent(pos, grid_size=2) - moves agent to randomly chosen valid neighbor, returns (new_pos, direction)
def move_agent(pos, grid_size=2):
    valid_moves = get_possible_moves(pos, grid_size)
    if not valid_moves:
        return pos, None
    direction = random.choice(valid_moves)
    new_pos = pos - 1 if direction == 'Left' else pos + 1
    return new_pos, direction

# 10. knowledge_all_clean(knowledge) - checks if agent believes all cells are clean (False if None or 1)
def knowledge_all_clean(knowledge):
    for status in knowledge:
        if status is None or status == 1:
            return False
    return True

# 11. all_clean(env) - checks true hidden environment state (for reference/debugging)
def all_clean(env):
    return all(cell == 0 for cell in env)

# 12. display_environment(knowledge, agent_pos, action, step) - prints current grid state and action
def display_environment(knowledge, agent_pos, action, step):
    """
    Formats the grid:
    D = Dirty (known)
    C = Clean (known)
    ? = Unknown
    A() = Agent present, perceiving
    A(V) = Agent vacuumed
    A(R) = Agent moving Right
    A(L) = Agent moving Left
    """
    header = f"Initial Stage:" if step == 0 else f"Run {step}:"
    cells_str = ""
    for i in range(len(knowledge)):
        status = knowledge[i]
        status_label = "?" if status is None else ("D" if status == 1 else "C")
        if i == agent_pos:
            if action == "perceive" or action is None:
                cell_repr = f"[A(),{status_label}]"
            elif action == "vacuum":
                cell_repr = f"[A(V),{status_label}]"
            elif action == "move_Right":
                cell_repr = f"[A(R),{status_label}]"
            elif action == "move_Left":
                cell_repr = f"[A(L),{status_label}]"
            else:
                cell_repr = f"[A({action}),{status_label}]"
        else:
            if status is None:
                cell_repr = "[?]"
            else:
                cell_repr = f"[{status_label}]"
        cells_str += cell_repr
    print(header)
    print(cells_str)

# 13. run_vacuum_world() - main driver function running the simulation loop
def run_vacuum_world(max_iterations=20, verbose=True):
    env = create_environment()
    env = place_dirt(env)
    agent_pos = place_agent(len(env))
    knowledge = create_knowledge(len(env))

    # Initial perception
    perceive(env, knowledge, agent_pos)
    step = 0
    display_environment(knowledge, agent_pos, "perceive", step)

    while not knowledge_all_clean(knowledge) and step < max_iterations:
        step += 1
        # If current cell is dirty in agent's knowledge, vacuum it first
        if knowledge[agent_pos] == 1:
            vacuum(env, knowledge, agent_pos)
            display_environment(knowledge, agent_pos, "vacuum", step)
        else:
            # Current cell is clean, decide to move
            new_pos, direction = move_agent(agent_pos, len(env))
            action_label = f"move_{direction}"
            display_environment(knowledge, agent_pos, action_label, step)
            # Move to new position and perceive
            agent_pos = new_pos
            perceive(env, knowledge, agent_pos)
            step += 1
            display_environment(knowledge, agent_pos, "perceive", step)

    if knowledge_all_clean(knowledge):
        print("All cells are clean! Task Completed.")
    else:
        print("Maximum iterations reached without full cleaning.")
    print(f"True environment state: {env} (All clean: {all_clean(env)})\n")

# Experiment 1: Vacuum Cleaner World Agent (2×1 Grid)

## 1. Overview & Objective
This experiment implements a **Function-Based (No Classes)** Intelligent Vacuum Cleaner World Agent operating in a $2 \times 1$ grid environment. 

### Key Characteristics:
- **Environment**: A $2 \times 1$ discrete grid where each cell is either Clean (`0`) or Dirty (`1`). This represents the ground truth, hidden state.
- **Partially Observable**: The agent does not have access to the full environment state initially. It maintains its own internal **knowledge array** initialized to unknown values (`None` / `?`).
- **Perception-Driven**: The agent only discovers whether a cell is clean or dirty when it physically visits and perceives that specific cell.
- **Rational Actions**:
  - If the current cell is **dirty**, the agent **must vacuum** it before it is permitted to move.
  - If the current cell is **clean**, the agent chooses a valid move (**Left** or **Right**) within grid boundaries.
- **Agent-Centric Termination**: Stopping condition strictly depends on the agent's *own knowledge* confirming all cells are clean (or reaching a maximum step threshold). The agent never peeks into the hidden environment to decide whether to stop.

"""

Design a Utility-Based Vacuum Cleaner Agent (m x n Grid) with Performance Metrics

import random
from collections import deque

# 1. create_environment(rows, cols)
def create_environment(rows, cols):
    """Creates a 2D matrix (rows x cols) filled with 0 (Clean)."""
    return [[0 for _ in range(cols)] for _ in range(rows)]

# 2. place_dirt(env, dirt_prob=0.4)
def place_dirt(env, dirt_prob=0.4):
    """Randomly sets cells to 1 (Dirty) based on dirt_prob."""
    rows, cols = len(env), len(env[0])
    dirt_count = 0
    for r in range(rows):
        for c in range(cols):
            if random.random() < dirt_prob:
                env[r][c] = 1
                dirt_count += 1
    # Ensure at least two dirty cells for meaningful evaluation
    if dirt_count == 0:
        env[0][0] = 1
        env[rows - 1][cols - 1] = 1
    return env

# 3. place_agent(rows, cols)
def place_agent(rows, cols):
    """Randomly places the agent at a starting (row, col) position."""
    return (random.randint(0, rows - 1), random.randint(0, cols - 1))

# 4. is_dirty(env, pos)
def is_dirty(env, pos):
    """Checks whether cell at (r, c) is dirty (1)."""
    r, c = pos
    return env[r][c] == 1

# 5. vacuum(env, pos)
def vacuum(env, pos):
    """Cleans the cell at position pos."""
    r, c = pos
    env[r][c] = 0
    return env

# 6. get_possible_moves(pos, rows, cols)
def get_possible_moves(pos, rows, cols):
    """
    Returns valid directional moves from pos: Up, Down, Left, Right
    as dictionary mapping action string to resulting (r, c).
    """
    r, c = pos
    moves = {}
    if r > 0:
        moves['Up'] = (r - 1, c)
    if r < rows - 1:
        moves['Down'] = (r + 1, c)
    if c > 0:
        moves['Left'] = (r, c - 1)
    if c < cols - 1:
        moves['Right'] = (r, c + 1)
    return moves

# 7. find_nearest_dirty_cell(env, pos, rows, cols)
def find_nearest_dirty_cell(env, pos, rows, cols):
    """
    Uses BFS shortest-path search over the grid to locate the nearest dirty cell
    and the distance to it. Returns (target_pos, distance).
    """
    if is_dirty(env, pos):
        return pos, 0

    queue = deque([(pos[0], pos[1], 0)])
    visited = {pos}

    while queue:
        r, c, dist = queue.popleft()
        if env[r][c] == 1:
            return (r, c), dist

        for nr, nc in [(r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)]:
            if 0 <= nr < rows and 0 <= nc < cols and (nr, nc) not in visited:
                visited.add((nr, nc))
                queue.append((nr, nc, dist + 1))

    return None, 0

# Helper: Count dirty cells within radius
def get_dirt_density_nearby(env, pos, rows, cols, radius=1):
    """Counts number of dirty cells within Chebyshev radius of pos."""
    r, c = pos
    density = 0
    for dr in range(-radius, radius + 1):
        for dc in range(-radius, radius + 1):
            nr, nc = r + dr, c + dc
            if 0 <= nr < rows and 0 <= nc < cols:
                if env[nr][nc] == 1:
                    density += 1
    return density

# 8. compute_utility(action, pos, env, rows, cols)
def compute_utility(action, pos, env, rows, cols):
    """
    Formula:
    U(a) = VacuumReward(a) - DistanceToNearestDirty(a) - MovementCost(a) + DirtDensityNearby(a)
    """
    if action == 'vacuum':
        resulting_pos = pos
        vacuum_reward = 10 if is_dirty(env, pos) else -5
        movement_cost = 0
    else:
        # action is a move: 'Up', 'Down', 'Left', 'Right'
        moves = get_possible_moves(pos, rows, cols)
        resulting_pos = moves[action]
        vacuum_reward = 0
        movement_cost = 1

    _, dist_to_dirty = find_nearest_dirty_cell(env, resulting_pos, rows, cols)
    density = get_dirt_density_nearby(env, resulting_pos, rows, cols, radius=1)

    utility = vacuum_reward - dist_to_dirty - movement_cost + density
    return utility

# 9. choose_best_action(pos, env, rows, cols)
def choose_best_action(pos, env, rows, cols):
    """
    Evaluates candidate actions (vacuum if dirty, plus all valid directional moves)
    and selects the action with maximum utility.
    """
    candidate_actions = []

    # Mandatory rule: If current cell is dirty, vacuuming is highly prioritized
    if is_dirty(env, pos):
        candidate_actions.append('vacuum')
    else:
        moves = get_possible_moves(pos, rows, cols)
        candidate_actions.extend(moves.keys())

    if not candidate_actions:
        return None

    best_action = None
    best_utility = float('-inf')

    for a in candidate_actions:
        u = compute_utility(a, pos, env, rows, cols)
        if u > best_utility:
            best_utility = u
            best_action = a

    return best_action, best_utility

# 10. all_clean(env)
def all_clean(env):
    """Checks if all cells in grid are clean (0)."""
    return all(cell == 0 for row in env for cell in row)

# 11. compute_theoretical_minimum(env, start_pos, rows, cols)
def compute_theoretical_minimum(env, start_pos, rows, cols):
    """
    Estimates minimum actions needed:
    Total dirty cells to vacuum + Manhattan distance or MST distance to visit all dirty cells.
    """
    dirty_cells = [(r, c) for r in range(rows) for c in range(cols) if env[r][c] == 1]
    if not dirty_cells:
        return 0
    # Minimum steps = number of vacuums + minimum path length
    # A simple lower bound is total dirt + max distance from start to any dirt
    max_dist = max(abs(start_pos[0] - r) + abs(start_pos[1] - c) for r, c in dirty_cells)
    return len(dirty_cells) + max_dist

# 12. update_metrics(metrics, action_taken)
def update_metrics(metrics, action_taken):
    """Updates performance metric tracking counters."""
    metrics['total_steps'] += 1
    if action_taken == 'vacuum':
        metrics['vacuums'] += 1
        metrics['energy_consumed'] += 2  # vacuuming costs 2 units
    else:
        metrics['moves'] += 1
        metrics['energy_consumed'] += 1  # moving costs 1 unit
    return metrics

# 13. display_environment(env, agent_pos, action, step, rows, cols)
def display_environment(env, agent_pos, action, step, rows, cols):
    """Displays the grid state with agent marker and cell statuses."""
    header = "Initial Stage:" if step == 0 else f"Step {step} (Action: {action}):"
    print(header)
    for r in range(rows):
        row_str = ""
        for c in range(cols):
            status = "D" if env[r][c] == 1 else "C"
            if (r, c) == agent_pos:
                if action == 'vacuum':
                    cell_repr = f"[A(V),{status}]"
                elif action in ['Up', 'Down', 'Left', 'Right']:
                    abbr = action[0]
                    cell_repr = f"[A({abbr}),{status}]"
                else:
                    cell_repr = f"[A(),{status}]"
            else:
                cell_repr = f"[{status}]"
            row_str += cell_repr
        print(row_str)
    print()

# 14. display_performance_summary(metrics)
def display_performance_summary(metrics):
    """Prints final summary of metrics and optimality ratio."""
    print("=" * 45)
    print("        PERFORMANCE SUMMARY REPORT")
    print("=" * 45)
    print(f"Total Steps Executed:      {metrics['total_steps']}")
    print(f"Total Clean/Vacuum Actions:{metrics['vacuums']}")
    print(f"Total Movement Steps:      {metrics['moves']}")
    print(f"Total Energy Consumed:     {metrics['energy_consumed']} units")
    print(f"Theoretical Min Steps:     {metrics['theoretical_min']}")
    if metrics['theoretical_min'] > 0:
        ratio = metrics['total_steps'] / metrics['theoretical_min']
        print(f"Optimality Ratio:          {ratio:.2f} (Actual / Theoretical)")
    print("=" * 45)

# 15. run_vacuum_world(rows, cols) - Fully Observable Mode
def run_vacuum_world(rows=3, cols=3, max_iterations=50):
    print(f"\n=======================================================")
    print(f" EXPERIMENT 2: Fully Observable Utility Vacuum ({rows}x{cols})")
    print(f"=======================================================\n")
    
    env = create_environment(rows, cols)
    env = place_dirt(env, dirt_prob=0.35)
    agent_pos = place_agent(rows, cols)

    metrics = {
        'total_steps': 0,
        'vacuums': 0,
        'moves': 0,
        'energy_consumed': 0,
        'theoretical_min': compute_theoretical_minimum(env, agent_pos, rows, cols)
    }

    display_environment(env, agent_pos, "None", 0, rows, cols)

    step = 0
    while not all_clean(env) and step < max_iterations:
        step += 1
        best_act, util = choose_best_action(agent_pos, env, rows, cols)

        if best_act == 'vacuum':
            vacuum(env, agent_pos)
            update_metrics(metrics, 'vacuum')
            display_environment(env, agent_pos, 'vacuum', step, rows, cols)
        else:
            moves = get_possible_moves(agent_pos, rows, cols)
            agent_pos = moves[best_act]
            update_metrics(metrics, best_act)
            display_environment(env, agent_pos, best_act, step, rows, cols)

    if all_clean(env):
        print("Success: All cells in grid are clean! Task Completed.")
    else:
        print("Stopped: Maximum iterations reached.")

    display_performance_summary(metrics)


# Partially Observable Utility Vacuum Extension (Belief State)
def run_partially_observable_utility_agent(rows=3, cols=3, max_iterations=60):
    print(f"\n=======================================================")
    print(f" Partially Observable Utility-Based Agent ({rows}x{cols})")
    print(f"=======================================================\n")

    env = create_environment(rows, cols)
    env = place_dirt(env, dirt_prob=0.35)
    agent_pos = place_agent(rows, cols)
    knowledge = [[None for _ in range(cols)] for _ in range(rows)]

    # Perceive starting cell
    r, c = agent_pos
    knowledge[r][c] = env[r][c]

    step = 0
    print(f"Initial State - Agent at {agent_pos}:")

    def belief_all_clean(k):
        return all(cell == 0 for row in k for cell in row if cell is not None) and \
               all(cell is not None for row in k for cell in row)

    while not belief_all_clean(knowledge) and step < max_iterations:
        step += 1
        r, c = agent_pos
        knowledge[r][c] = env[r][c]

        if knowledge[r][c] == 1:
            vacuum(env, agent_pos)
            knowledge[r][c] = 0
            action = 'vacuum'
        else:
            # Move towards nearest unvisited cell or known dirty cell
            moves = get_possible_moves(agent_pos, rows, cols)
            # Preference: move to unknown cells ('?') or dirty cells ('D')
            best_move = None
            best_score = float('-inf')
            for act, npos in moves.items():
                nr, nc = npos
                score = 0
                if knowledge[nr][nc] is None:
                    score += 5  # Exploration reward
                elif knowledge[nr][nc] == 1:
                    score += 10 # Known dirt
                else:
                    score -= 1  # Already clean
                if score > best_score:
                    best_score = score
                    best_move = act

def get_path_cost(adj, path):
    """Calculates the total edge cost of a given path."""
    cost = 0
    for u, v in zip(path[:-1], path[1:]):
        for nbr, w in adj[u]:
            if nbr == v:
                cost += w
                break
    return cost

def bfs_path(adj, start='A', goal='F'):
    """
    Breadth-First Search (BFS): ignores weights during search.
    Returns (path, total_cost, visit_order).
    """
    queue = deque([[start]])
    visited = {start}
    visit_order = [start]

    while queue:
        path = queue.popleft()
        node = path[-1]
        if node == goal:
            return path, get_path_cost(adj, path), visit_order

        for nbr, _ in sorted(adj[node], key=lambda x: x[0]):
            if nbr not in visited:
                visited.add(nbr)
                visit_order.append(nbr)
                queue.append(path + [nbr])

    return None, 0, visit_order

def dfs_path(adj, start='A', goal='F'):
    """
    Depth-First Search (DFS): ignores weights during search.
    Returns (path, total_cost, visit_order).
    """
    stack = [[start]]
    visited = set()
    visit_order = []

    while stack:
        path = stack.pop()
        node = path[-1]
        if node not in visited:
            visited.add(node)
            visit_order.append(node)
            if node == goal:
                return path, get_path_cost(adj, path), visit_order

            # Explore neighbors in alphabetical order
            for nbr, _ in sorted(adj[node], key=lambda x: x[0], reverse=True):
                if nbr not in visited:
                    stack.append(path + [nbr])

    return None, 0, visit_order

def dijkstra_path(adj, start='A', goal='F'):
    """
    Dijkstra's Algorithm: uses heapq, dist, and parent dictionaries.
    Returns (path, total_cost, visit_order).
    """
    dist = {n: math.inf for n in adj}
    parent = {n: None for n in adj}
    dist[start] = 0
    pq = [(0, start)]
    visit_order = []
    visited = set()

    while pq:
        d, u = heapq.heappop(pq)
        if u in visited:
            continue
        visited.add(u)
        visit_order.append(u)
        if u == goal:
            break

        for v, weight in sorted(adj[u], key=lambda x: x[0]):
            if dist[u] + weight < dist[v]:
                dist[v] = dist[u] + weight
                parent[v] = u
                heapq.heappush(pq, (dist[v], v))

    # Reconstruct shortest path
    path = []
    curr = goal
    while curr is not None:
        path.append(curr)
        curr = parent[curr]
    path.reverse()

    return path, dist[goal], visit_order

## 2. Theoretical Note: Why 0 is Wrong as a "No Edge" Sentinel
In unweighted graphs, `0` is often used to denote absence of an edge and `1` denotes presence. However, in **weighted graphs**, `0` is a valid edge weight representing a **zero-cost traversal** between two nodes.

If `0` is used to represent "no edge":
- Shortest-path algorithms like Dijkstra's algorithm would interpret `matrix[u][v] == 0` as a free edge between `u` and `v`.
- As a consequence, disconnected nodes would appear to have zero-distance paths between each other, leading to completely corrupted search paths and distance calculations.
- Hence, the sentinel `math.inf` (infinity) is mathematically and practically mandatory, representing infinite impedance or absence of a path.

def get_path_cost(adj, path):
    """Calculates total path cost."""
    cost = 0
    for u, v in zip(path[:-1], path[1:]):
        for nbr, w in adj[u]:
            if nbr == v:
                cost += w
                break
    return cost

# 1. best_first_path(adj_list, h, 'A', 'F') - Greedy Best-First Search
def best_first_path(adj, h_dict, start='A', goal='F'):
    """
    Greedy Best-First Search:
    Expands nodes based strictly on heuristic value h(n).
    Returns (path, total_cost, visit_order).
    """
    # Priority Queue elements: (heuristic_cost, current_node, path)
    pq = [(h_dict[start], start, [start])]
    visited = set()
    visit_order = []

    while pq:
        h_val, u, path = heapq.heappop(pq)
        if u in visited:
            continue
        visited.add(u)
        visit_order.append(u)

        if u == goal:
            return path, get_path_cost(adj, path), visit_order

        for v, _ in sorted(adj[u], key=lambda x: x[0]):
            if v not in visited:
                heapq.heappush(pq, (h_dict[v], v, path + [v]))

    return None, 0, visit_order

# 2. astar_path(adj_list, h, 'A', 'F') - A* Search
def astar_path(adj, h_dict, start='A', goal='F'):
    """
    A* Search:
    Expands nodes based on f(n) = g(n) + h(n), where g(n) is path cost from start.
    Returns (path, total_cost, visit_order).
    """
    # Priority Queue elements: (f_score, g_score, current_node, path)
    pq = [(h_dict[start], 0, start, [start])]
    g_score = {node: math.inf for node in adj}
    g_score[start] = 0
    visited = set()
    visit_order = []

    while pq:
        f_val, g_val, u, path = heapq.heappop(pq)
        if u in visited:
            continue
        visited.add(u)
        visit_order.append(u)

        if u == goal:
            return path, g_val, visit_order

        for v, weight in sorted(adj[u], key=lambda x: x[0]):
            tentative_g = g_val + weight
            if tentative_g < g_score[v]:
                g_score[v] = tentative_g
                f_val_v = tentative_g + h_dict[v]
                heapq.heappush(pq, (f_val_v, tentative_g, v, path + [v]))

    return None, 0, visit_order

def hill_climbing_continuous(x_init=1.5, step_size=0.05, max_iter=200):
    """
    Standard Steepest-Ascent / Local Hill Climbing for continuous domain.
    Returns (best_x, best_f, history).
    """
    current_x = x_init
    current_f = objective_function(current_x)
    history = [(current_x, current_f)]

    for _ in range(max_iter):
        cand_left = max(0.0, min(10.0, current_x - step_size))
        cand_right = max(0.0, min(10.0, current_x + step_size))

        f_left = objective_function(cand_left)
        f_right = objective_function(cand_right)

        if f_left > current_f and f_left >= f_right:
            current_x, current_f = cand_left, f_left
            history.append((current_x, current_f))
        elif f_right > current_f:
            current_x, current_f = cand_right, f_right
            history.append((current_x, current_f))
        else:
            # Reached a local peak (gradient is non-positive in both directions)
            break

    return current_x, current_f, history

def simulated_annealing_continuous(x_init=1.5, t_init=50.0, alpha=0.995, max_iter=2000):
    """
    Simulated Annealing on 1D continuous landscape.
    Accepts worse moves probabilistically using P = exp(delta / T) to escape local optima.
    Returns (best_x, best_f, history).
    """
    current_x = x_init
    current_f = objective_function(current_x)
    best_x, best_f = current_x, current_f
    history = [(current_x, current_f)]

    t = t_init
    for _ in range(max_iter):
        step = random.uniform(-4.0, 4.0)
        cand_x = max(0.0, min(10.0, current_x + step))
        cand_f = objective_function(cand_x)

        delta = cand_f - current_f

        # Metropolis acceptance criterion
        if delta > 0 or math.exp(delta / max(t, 1e-8)) > random.random():
            current_x, current_f = cand_x, cand_f
            history.append((current_x, current_f))
            if current_f > best_f:
                best_x, best_f = current_x, current_f

        t *= alpha
        if t < 1e-4:
            break

    return best_x, best_f, history

def get_8queens_score(state):
    """
    Objective function: Number of non-attacking queen pairs.
    Maximum possible score = C(8, 2) = 28 (zero conflicts).
    """
    conflicts = 0
    n = len(state)
    for i in range(n):
        for j in range(i + 1, n):
            if state[i] == state[j] or abs(state[i] - state[j]) == abs(i - j):
                conflicts += 1
    return 28 - conflicts

def get_random_neighbor_8queens(state):

    neighbor[col] = random.choice(available_rows)

## 2. Theoretical Principles
### Simulated Annealing Acceptance Criterion:
When considering a transition from state $s$ to neighbor $s'$ with energy difference $\Delta E = f(s') - f(s)$:
- If $\Delta E > 0$ (improving move): Accept unconditionally.
- If $\Delta E \le 0$ (worse move): Accept with probability:
  $$P(\text{accept}) = \exp\left(\frac{\Delta E}{T}\right)$$
  As temperature $T$ gradually decreases according to cooling schedule $T_{k+1} = \alpha T_k$, the probability of accepting downhill moves drops to 0, smoothly transitioning from random exploration to greedy refinement.

    elif state == human_player:

## 2. Minimax Mathematical Formulation
For a game state $s$, the minimax value is defined recursively:
$$\text{Minimax}(s) = \begin{cases} 
\text{Utility}(s) & \text{if } \text{Terminal}(s) \\
\max_{a \in \text{Actions}(s)} \text{Minimax}(\text{Result}(s, a)) & \text{if } \text{Player}(s) = \text{MAX (AI)} \\
\min_{a \in \text{Actions}(s)} \text{Minimax}(\text{Result}(s, a)) & \text{if } \text{Player}(s) = \text{MIN (Human)}
\end{cases}$$

With depth penalty:
- AI Win: $+10 - \text{depth}$
- Human Win: $\text{depth} - 10$
- Draw: $0$

# Experiment 7: Connect-3 on a 4×4 Board (Minimax vs. Alpha–Beta Pruning)

## 1. Overview & Objective
This experiment implements the game of **Connect-3** played on a $4 \times 4$ board (16 cells total) under two search algorithms:
1. **Plain Minimax Search**: Exhaustively visits game tree states.
2. **Minimax with Alpha–Beta Pruning**: Prunes branches that cannot influence the final decision, dramatically cutting down the state space.

### Game Rules:
- **Board**: 4 columns $\times$ 4 rows (16 cells). Empty cells are represented by `-`.
- **Turns**: Two players, Human (`X`) and AI (`O`), move alternately. `X` moves first.
- **Gravity**: A move is a choice of column $[1, 4]$. The piece drops to the lowest unoccupied cell in that column. Full columns are invalid.
- **Winning Condition**: First player to form an uninterrupted line of 3 identical pieces horizontally, vertically, or diagonally wins immediately.
- **Draw**: All 16 cells are filled with no player achieving a 3-in-a-row.

---

## 2. Theoretical Principles: Alpha–Beta Pruning
Alpha–Beta pruning maintains two bounds during recursive exploration:
- $\alpha$: The best (maximum) value that the MAX player is guaranteed so far.
- $\beta$: The best (minimum) value that the MIN player is guaranteed so far.

Whenever a branch yields a condition where:
$$\beta \le \alpha$$
The current node's parent would never select this path. The remaining child branches are **pruned**, saving enormous numbers of recursive function calls without any loss in optimality.

class WumpusWorld:
    def __init__(self, n=4, wumpus_pos=None, pits=None, gold_pos=None, p=0.2, seed=None):
        self.n = n
        if seed is not None:
            random.seed(seed)

        all_cells = [(x, y) for x in range(1, n + 1) for y in range(1, n + 1) if (x, y) != (1, 1)]

        # Wumpus Placement
        if wumpus_pos is not None:
            self.wumpus_pos = wumpus_pos
        else:
            self.wumpus_pos = random.choice(all_cells)

        # Pit Placement
        if pits is not None:
            self.pits = set(pits)
        else:
            self.pits = {c for c in all_cells if random.random() < p}

        # Gold Placement (No pit and no Wumpus)
        if gold_pos is not None:
            self.gold_pos = gold_pos
        else:
            valid_gold = [c for c in all_cells if c not in self.pits and c != self.wumpus_pos]
            self.gold_pos = random.choice(valid_gold) if valid_gold else (n, n)

        self.wumpus_alive = True

    def get_neighbors(self, pos):
        x, y = pos
        nbrs = []
        # Move order preference: RIGHT, UP, LEFT, DOWN
        for dx, dy in [(1, 0), (0, 1), (-1, 0), (0, -1)]:
            nx, ny = x + dx, y + dy
            if 1 <= nx <= self.n and 1 <= ny <= self.n:
                nbrs.append((nx, ny))
        return nbrs

    def sense_percepts(self, pos, just_screamed=False):
        nbrs = self.get_neighbors(pos)
        stench = (self.wumpus_pos in nbrs)  # Dead wumpus still smells
        breeze = any(p in nbrs for p in self.pits)
        glitter = (pos == self.gold_pos)
        percepts = []
        if stench: percepts.append("Stench")
        if breeze: percepts.append("Breeze")
        if glitter: percepts.append("Glitter")
        if just_screamed: percepts.append("Scream")
        return percepts

    def shoot_arrow(self, pos, direction):
        """Fires single arrow from pos along direction. Returns True if Wumpus is killed."""
        if not self.wumpus_alive:
            return False

        cx, cy = pos
        while True:
            if direction == 'RIGHT': cx += 1
            elif direction == 'LEFT': cx -= 1
            elif direction == 'UP': cy += 1
            elif direction == 'DOWN': cy -= 1

            if 1 <= cx <= self.n and 1 <= cy <= self.n:
                if (cx, cy) == self.wumpus_pos:
                    self.wumpus_alive = False
                    return True
            else:
                break
        return False

class RuleBasedAgent:
    def __init__(self, n=4):
        self.n = n
        self.pos = (1, 1)
        self.visited = set()
        self.pit_free = {(1, 1)}
        self.all_cells = {(x, y) for x in range(1, n + 1) for y in range(1, n + 1)}
        self.wumpus_candidates = self.all_cells - {(1, 1)}
        self.wumpus_dead = False
        self.arrow_used = False
        self.actions_log = []
        self.trace = []

    def get_neighbors(self, pos):
        x, y = pos
        nbrs = []
        for dx, dy in [(1, 0), (0, 1), (-1, 0), (0, -1)]:
            nx, ny = x + dx, y + dy
            if 1 <= nx <= self.n and 1 <= ny <= self.n:
                nbrs.append((nx, ny))
        return nbrs

    def is_safe(self, cell):
        """A cell is safe iff it is in PitFree and (WumpusDead or not a WumpusCandidate)."""
        return (cell in self.pit_free) and (self.wumpus_dead or (cell not in self.wumpus_candidates))

    def update_beliefs(self, percepts):
        self.visited.add(self.pos)

        # 1. PitFree Update: If no breeze, all neighbors are pit free
        if "Breeze" not in percepts:
            for nbr in self.get_neighbors(self.pos):
                self.pit_free.add(nbr)

        # 2. WumpusCandidates Update
        nbrs = set(self.get_neighbors(self.pos))
        if "Stench" in percepts:
            if not self.wumpus_dead:
                self.wumpus_candidates &= nbrs
        else:
            self.wumpus_candidates -= nbrs
            self.wumpus_candidates.discard(self.pos)

    def find_path_to_target(self, target):
        """BFS shortest path through visited/safe cells from current pos to target."""
        queue = deque([(self.pos, [self.pos])])
        visited_bfs = {self.pos}
        while queue:
            curr, path = queue.popleft()
            if curr == target:
                return path
            for nbr in self.get_neighbors(curr):
                if (nbr in self.visited or self.is_safe(nbr)) and nbr not in visited_bfs:
                    visited_bfs.add(nbr)
                    queue.append((nbr, path + [nbr]))
        return None

    def run(self, world):
        step = 0
        just_screamed = False

        while True:
            step += 1
            percepts = world.sense_percepts(self.pos, just_screamed)
            just_screamed = False
            self.update_beliefs(percepts)

            safe_cells = sorted([c for c in self.all_cells if self.is_safe(c)])

            # 1. Glitter Check: GRAB and stop
            if "Glitter" in percepts:
                self.actions_log.append("GRAB")
                self.trace.append({
                    'step': step,
                    'action': 'GRAB',
                    'pos': self.pos,
                    'percepts': percepts,
                    'safe_cells': safe_cells
                })
                shortest_path = self.find_path_to_target((1, 1))
                if shortest_path:
                    shortest_path.reverse()
                return "POSSIBLE", self.actions_log, shortest_path, self.trace

            # 2. Search for nearest unvisited safe cell
            unvisited_safe = [c for c in self.all_cells if self.is_safe(c) and c not in self.visited]

            if unvisited_safe:
                # Find closest unvisited safe cell via BFS
                best_path = None
                for candidate in unvisited_safe:
                    path = self.find_path_to_target(candidate)
                    if path:
                        if best_path is None or len(path) < len(best_path):
                            best_path = path

                if best_path and len(best_path) > 1:
                    next_pos = best_path[1]
                    dx = next_pos[0] - self.pos[0]
                    dy = next_pos[1] - self.pos[1]
                    act = "RIGHT" if dx == 1 else ("LEFT" if dx == -1 else ("UP" if dy == 1 else "DOWN"))
                    self.actions_log.append(act)
                    self.trace.append({
                        'step': step,
                        'action': f"MOVE {act} -> {next_pos}",
                        'pos': self.pos,
                        'percepts': percepts,
                        'safe_cells': safe_cells
                    })
                    self.pos = next_pos
                    continue

            # 3. If no safe unvisited cell and arrow unused, shoot along candidate line
            if not self.arrow_used and self.wumpus_candidates:
                self.arrow_used = True
                shot_direction = None

                # Prioritize adjacent direction containing candidate (RIGHT, UP, LEFT, DOWN)
                for direction in ['RIGHT', 'UP', 'LEFT', 'DOWN']:
                    cx, cy = self.pos
                    if direction == 'RIGHT': target = (cx + 1, cy)
                    elif direction == 'LEFT': target = (cx - 1, cy)
                    elif direction == 'UP': target = (cx, cy + 1)
                    elif direction == 'DOWN': target = (cx, cy - 1)
                    if target in self.wumpus_candidates:
                        shot_direction = direction
                        break

                # Otherwise find any candidate along row or column
                if shot_direction is None:
                    sorted_cands = sorted(list(self.wumpus_candidates), key=lambda c: (abs(c[0]-self.pos[0]) + abs(c[1]-self.pos[1])))
                    for cand in sorted_cands:
                        if cand[1] == self.pos[1]:
                            shot_direction = "RIGHT" if cand[0] > self.pos[0] else "LEFT"
                            break
                        elif cand[0] == self.pos[0]:
                            shot_direction = "UP" if cand[1] > self.pos[1] else "DOWN"
                            break

                if shot_direction:
                    hit = world.shoot_arrow(self.pos, shot_direction)
                    act_label = f"SHOOT_{shot_direction}"
                    if hit:
                        self.wumpus_dead = True
                        self.wumpus_candidates.clear()
                        just_screamed = True
                        act_label += " (Scream)"
                    else:
                        cx, cy = self.pos
                        while True:
                            if shot_direction == "RIGHT": cx += 1
                            elif shot_direction == "LEFT": cx -= 1
                            elif shot_direction == "UP": cy += 1
                            elif shot_direction == "DOWN": cy -= 1
                            if 1 <= cx <= self.n and 1 <= cy <= self.n:
                                self.wumpus_candidates.discard((cx, cy))
                            else:
                                break

                    self.actions_log.append(act_label)
                    self.trace.append({
                        'step': step,
                        'action': act_label,
                        'pos': self.pos,
                        'percepts': percepts,
                        'safe_cells': safe_cells
                    })
                    continue

            # 4. If still no safe unvisited cells, report NOT POSSIBLE
            self.trace.append({
                'step': step,
                'action': 'TERMINATE (No safe moves)',
                'pos': self.pos,
                'percepts': percepts,
                'safe_cells': safe_cells
            })
            return "NOT POSSIBLE", self.actions_log, None, self.trace

## 1. Overview & Objective
This experiment implements a **Knowledge-Based Rule Agent** exploring the **Wumpus World** without any initial map knowledge:
- **Environment**: $N \times N$ discrete grid (default $4 \times 4$), 1-indexed coordinates $(x, y)$ where $(1, 1)$ is the bottom-left starting cell.

  - One Wumpus (never at $(1, 1)$), can co-exist with a pit.

  - `Glitter`: Gold in current cell.
  - `Scream`: Wumpus was just killed by arrow.
- **Safety Criterion**:
  - A cell is safe **iff** it is in `PitFree` **and** (`WumpusDead` or not in `WumpusCandidates`).
  - Killing the Wumpus does **not** make its cell pit-free!

## 2. Rule-Based Deduction Logic
1. **`PitFree` Tracking**:
   - Initialized to `{(1, 1)}`.
   - If current cell has **no Breeze**, all of its adjacent neighbors are added to `PitFree`.
2. **`WumpusCandidates` Tracking**:
   - Initialized to all cells except $(1, 1)$.
   - If current cell has **no Stench**, remove cell and its neighbors from `WumpusCandidates`.
   - If current cell has **Stench**, intersect `WumpusCandidates` with neighbors of current cell.
   - If arrow flies through a line of cells without triggering a `Scream`, those cells are removed from `WumpusCandidates`.
   - A `Scream` sets `WumpusDead = True`.
3. **Navigation**:
   - When multiple safe cells exist, the agent uses Breadth-First Search (BFS) through visited/safe cells to plan the shortest safe route.
   - If no safe cell is reachable and arrow is available, shoots towards candidate cells.
   - If no safe cell can be proven, terminates reporting `NOT POSSIBLE`.

def pl_true(sentence, model):
    """
    Evaluates a propositional sentence in a given model (dict of symbol -> bool).
    Representations:
      'P'                       -> string symbol
      ('not', s)                -> negation
      ('and', s1, s2, ...)      -> conjunction
      ('or', s1, s2, ...)       -> disjunction
      ('implies', p, q)         -> material implication (not p or q)
      ('iff', p, q)             -> biconditional (p == q)
    """
    if isinstance(sentence, bool):
        return sentence
    if isinstance(sentence, str):
        if sentence in model:
            return model[sentence]
        raise KeyError(f"Symbol '{sentence}' not assigned in model: {model}")

    op = sentence[0]
    args = sentence[1:]

    if op == 'not':
        return not pl_true(args[0], model)
    elif op == 'and':
        return all(pl_true(a, model) for a in args)
    elif op == 'or':
        return any(pl_true(a, model) for a in args)
    elif op == 'implies':
        return (not pl_true(args[0], model)) or pl_true(args[1], model)
    elif op == 'iff':
        return pl_true(args[0], model) == pl_true(args[1], model)
    else:
        raise ValueError(f"Unknown operator: {op}")

def get_symbols(sentence):
    """Extracts all propositional variable names from a sentence."""
    if isinstance(sentence, str):
        return {sentence}
    if isinstance(sentence, (list, tuple)):
        symbols = set()
        for elem in sentence[1:]:
            symbols |= get_symbols(elem)
        return symbols
    return set()

def sentence_to_str(s):
    """Pretty prints nested tuple propositional sentences."""
    if isinstance(s, str):
        return s
    op = s[0]
    if op == 'not':
        return f"~{sentence_to_str(s[1])}"
    elif op == 'and':
        return "(" + " & ".join(sentence_to_str(a) for a in s[1:]) + ")"
    elif op == 'or':
        return "(" + " | ".join(sentence_to_str(a) for a in s[1:]) + ")"
    elif op == 'implies':
        return f"({sentence_to_str(s[1])} => {sentence_to_str(s[2])})"
    elif op == 'iff':
        return f"({sentence_to_str(s[1])} <=> {sentence_to_str(s[2])})"
    return str(s)

def print_truth_table(sentence):
    """Generates and prints the complete truth table for any propositional sentence."""
    symbols = sorted(list(get_symbols(sentence)))
    n = len(symbols)
    expr_str = sentence_to_str(sentence)

    header = " | ".join(f"{s:^5}" for s in symbols) + f" | {expr_str:^25}"
    separator = "-" * len(header)

    print(f"\nTruth Table for: {expr_str}")
    print(separator)
    print(header)
    print(separator)

    for values in itertools.product([True, False], repeat=n):
        model = dict(zip(symbols, values))
        res = pl_true(sentence, model)
        row_str = " | ".join(f"{'T' if v else 'F':^5}" for v in values) + f" | {'T' if res else 'F':^25}"
        print(row_str)
    print(separator + "\n")

def tt_entails(kb, alpha):
    """
    Checks KB |= alpha by truth-table enumeration:
    alpha must be True in every model where KB is True.
    """
    if isinstance(kb, (list, set)):
        kb_sentence = ('and', *kb) if len(kb) > 1 else (list(kb)[0] if kb else True)
    else:
        kb_sentence = kb

    symbols = sorted(list(get_symbols(kb_sentence) | get_symbols(alpha)))

    for values in itertools.product([True, False], repeat=len(symbols)):
        model = dict(zip(symbols, values))
        if pl_true(kb_sentence, model):
            if not pl_true(alpha, model):
                return False
    return True

def is_valid(sentence):
    """A sentence is valid (tautology) if it is True in all 2^n models."""
    symbols = sorted(list(get_symbols(sentence)))
    for values in itertools.product([True, False], repeat=len(symbols)):
        model = dict(zip(symbols, values))
        if not pl_true(sentence, model):
            return False
    return True

def is_satisfiable(sentence):
    """A sentence is satisfiable if it is True in at least one model."""
    symbols = sorted(list(get_symbols(sentence)))
    for values in itertools.product([True, False], repeat=len(symbols)):
        model = dict(zip(symbols, values))
        if pl_true(sentence, model):
            return True
    return False

class LogicWumpusAgent:
    """
    Solves Wumpus World navigation using Propositional Logic Model Checking.
    Proves cell safety by querying:
      KB |= ~P_x_y   (provably Pit-free)
      KB |= ~W_x_y   (provably Wumpus-free)
    """
    def __init__(self, n=4):
        self.n = n
        self.pos = (1, 1)
        self.visited = set()
        self.proven_safe = {(1, 1)}
        self.pit_clauses = {}      # (x, y) -> list of breeze clauses
        self.wumpus_clauses = {}   # (x, y) -> list of stench clauses

    def get_neighbors(self, pos):
        x, y = pos
        nbrs = []
        for dx, dy in [(1, 0), (0, 1), (-1, 0), (0, -1)]:
            nx, ny = x + dx, y + dy
            if 1 <= nx <= self.n and 1 <= ny <= self.n:
                nbrs.append((nx, ny))
        return nbrs

    def record_percepts(self, pos, percepts):
        x, y = pos
        self.visited.add(pos)
        nbrs = self.get_neighbors(pos)

        breeze_sym = f"B_{x}_{y}"
        pit_nbrs = [f"P_{nx}_{ny}" for nx, ny in nbrs]
        b_equiv = ('iff', breeze_sym, ('or', *pit_nbrs))
        b_fact = breeze_sym if "Breeze" in percepts else ('not', breeze_sym)
        self.pit_clauses[pos] = [b_equiv, b_fact]

        stench_sym = f"S_{x}_{y}"
        w_nbrs = [f"W_{nx}_{ny}" for nx, ny in nbrs]
        s_equiv = ('iff', stench_sym, ('or', *w_nbrs))
        s_fact = stench_sym if "Stench" in percepts else ('not', stench_sym)
        self.wumpus_clauses[pos] = [s_equiv, s_fact]

    def is_cell_provably_safe(self, target):
        if target in self.proven_safe:
            return True
        tx, ty = target
        p_target = f"P_{tx}_{ty}"
        w_target = f"W_{tx}_{ty}"

        # Collect clauses from visited neighbors of target
        relevant_pit_kb = []
        relevant_wumpus_kb = []
        for nbr in self.get_neighbors(target):
            if nbr in self.pit_clauses:
                relevant_pit_kb.extend(self.pit_clauses[nbr])
            if nbr in self.wumpus_clauses:
                relevant_wumpus_kb.extend(self.wumpus_clauses[nbr])

        if not relevant_pit_kb or not relevant_wumpus_kb:
            return False

        # Model checking for target
        pit_free = tt_entails(relevant_pit_kb, ('not', p_target))
        wumpus_free = tt_entails(relevant_wumpus_kb, ('not', w_target))

        if pit_free and wumpus_free:
            self.proven_safe.add(target)
            return True
        return False

    def find_safe_path_to_gold(self, gold_pos, wumpus_pos, pits, max_steps=25):
        current = (1, 1)
        path_taken = [current]
        step = 0

        while current != gold_pos and step < max_steps:
            step += 1
            nbrs = self.get_neighbors(current)
            stench = (wumpus_pos in nbrs)
            breeze = any(p in nbrs for p in pits)
            percepts = []
            if stench: percepts.append("Stench")
            if breeze: percepts.append("Breeze")

            self.record_percepts(current, percepts)

            # Deduce safety of adjacent cells using propositional logic
            for nbr in nbrs:
                self.is_cell_provably_safe(nbr)

            unvisited_safe = [c for c in self.proven_safe if c not in self.visited]
            if not unvisited_safe:
                break

            # BFS through proven safe cells to nearest unvisited safe cell
            queue = deque([(current, [current])])
            seen = {current}
            next_step = None

            while queue:
                node, path = queue.popleft()
                if node in unvisited_safe:
                    if len(path) > 1:
                        next_step = path[1]
                    break
                for n in self.get_neighbors(node):
                    if n in self.proven_safe and n not in seen:
                        seen.add(n)
                        queue.append((n, path + [n]))

            if next_step:
                current = next_step
                path_taken.append(current)
            else:
                break

        return path_taken

## 1. Overview & Objective
This experiment implements a **Propositional Logic Inference Engine** using model checking (truth-table enumeration) and applies it to **Wumpus World reasoning**:
1. **Sentence Representation**: Represent complex logical expressions using nested Python tuples:
   - Literal: `'P'`
   - Negation: `('not', 'P')`
   - Conjunction: `('and', 'P', 'Q', 'R')`
   - Disjunction: `('or', 'P', 'Q')`
   - Implication: `('implies', 'P', 'Q')`
   - Biconditional: `('iff', 'P', 'Q')`
2. **Model Checking & Inference**:
   - `pl_true(sentence, model)`: Recursively evaluates the truth value of any sentence under a given assignment dictionary.
   - `print_truth_table(sentence)`: Computes and prints the full truth table over all $2^n$ models.
   - `tt_entails(KB, alpha)`: Verifies whether $KB \models \alpha$ by proving that $\alpha$ holds in every model satisfying $KB$.
   - `is_valid(sentence)`: Checks if a sentence is a Tautology (true in all $2^n$ models).
   - `is_satisfiable(sentence)`: Checks if a sentence is true in at least one model.
3. **Application to Wumpus World**:
   - Encodes physical axioms connecting percepts to world hazards:
     $$B_{x,y} \iff \bigvee_{(nx, ny) \in \text{Neighbors}(x, y)} P_{nx, ny}$$
     $$S_{x,y} \iff \bigvee_{(nx, ny) \in \text{Neighbors}(x, y)} W_{nx, ny}$$
   - Formally proves a cell $(nx, ny)$ is safe by verifying $KB \models \neg P_{nx, ny}$ and $KB \models \neg W_{nx, ny}$.

   - $(P \implies Q) \iff (\neg Q \implies \neg P)$
